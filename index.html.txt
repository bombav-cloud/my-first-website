<script>
const ROOMS=[
 {id:"gab",name:"Grand Aran Ballroom",sub:"(GAB)",full:"ห้องประชุม Grand Aran Ballroom (GAB)"},
 {id:"dir",name:"ห้องประชุมผู้อำนวยการ",sub:"",full:"ห้องประชุมผู้อำนวยการ"},
 {id:"adm",name:"ห้องประชุมกลุ่มงานบริหาร",sub:"",full:"ห้องประชุมกลุ่มงานบริหาร"}
];
let all=[],col=null,viewDate=null,idle=null;
const $=id=>document.getElementById(id);
const pad=n=>String(n).padStart(2,"0");
const key=d=>d.getFullYear()+"-"+pad(d.getMonth()+1)+"-"+pad(d.getDate());
const esc=s=>String(s).replace(/[&<>"]/g,c=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;"}[c]));
function view(){const h=(location.hash||"").slice(1);return ROOMS.find(r=>r.id===h)?h:"all"}
function render(){
  const now=new Date(),real=key(now),today=viewDate||real,isToday=today===real,hm=pad(now.getHours())+":"+pad(now.getMinutes()),v=view();
  $("clock").textContent=hm;
  const [yy,mm,dd]=today.split("-").map(Number);$("date").textContent=new Date(yy,mm-1,dd).toLocaleDateString("th-TH",{weekday:"long",day:"numeric",month:"long",year:"numeric"});$("pk").value=today;$("td").style.visibility=isToday?"hidden":"visible";
  const rs=v==="all"?ROOMS:ROOMS.filter(r=>r.id===v);
  
  const box=$("rooms");box.className="rooms "+(v==="all"?"all":"single");
  box.innerHTML=rs.map(r=>{
    const ms=all.filter(m=>m.date===today&&m.room===r.id).sort((a,b)=>a.start.localeCompare(b.start));
    const t=isToday?hm:(today<real?"24:00":"00:00");const cur=ms.find(m=>t>=m.start&&t<m.end),nxt=ms.find(m=>m.start>t);
    const body=ms.length?ms.map(m=>{
      let c="soon";if(t>=m.end)c="done";else if(t>=m.start)c="now";
      return `<div class="m ${c}"><div class="t">${m.start} - ${m.end}</div><div class="s">${esc(m.topic)}</div></div>`;
    }).join(""):`<div class="none">ไม่มีการประชุม</div>`;
    let a,b="";const pl=!isToday;
    if(pl){a=ms.length?ms.length+" รายการ":"ไม่มีประชุม"}else if(cur){a="ไม่ว่าง";b=`กำลังประชุม<br>ถึง ${cur.end} น.`}
    else{a="ว่าง";b=nxt?`ประชุมถัดไป<br>${nxt.start} น.`:"ว่างตลอดวัน"}
    return `<section class="room"><div class="rn">${r.name}<small>${r.sub}</small></div><div class="list">${body}</div><div class="bar ${pl?"plain":cur?"busy":""}"><div class="a">${a}</div><div class="b">${b}</div></div></section>`;
  }).join("");
  const up=all.filter(m=>m.date>=real).sort((a,b)=>(a.date+a.start).localeCompare(b.date+b.start));
  $("up").innerHTML=up.length?up.map(m=>`<div class="row"><span>${m.date} ${m.start}-${m.end} · ${ROOMS.find(r=>r.id===m.room)?.name||""} · ${esc(m.topic)}</span><button data-id="${m.id}">ลบ</button></div>`).join(""):`<div class="row" style="color:var(--mute)">ยังไม่มีรายการ</div>`;
}
$("fr").innerHTML=ROOMS.map(r=>`<option value="${r.id}">${r.full}</option>`).join("");
$("fd").value=key(new Date());
function go(d){viewDate=d===key(new Date())?null:d;clearTimeout(idle);if(viewDate)idle=setTimeout(()=>{viewDate=null;render()},90000);render()}
function shift(n){const b=viewDate||key(new Date()),[y,m,d]=b.split("-").map(Number);go(key(new Date(y,m-1,d+n)))}
$("pv").onclick=()=>shift(-1);$("nx").onclick=()=>shift(1);$("td").onclick=()=>go(key(new Date()));
$("pk").onchange=e=>{if(e.target.value)go(e.target.value)};
addEventListener("keydown",e=>{if(e.target.tagName==="INPUT")return;if(e.key==="ArrowLeft")shift(-1);if(e.key==="ArrowRight")shift(1)});
$("tg").onclick=()=>$("admin").classList.add("on");
$("cl").onclick=()=>$("admin").classList.remove("on");
window.addEventListener("hashchange",()=>{render()});
$("add").onclick=async()=>{
  const m={date:$("fd").value,room:$("fr").value,start:$("fs").value,end:$("fe").value,topic:$("ft").value.trim()};
  const msg=$("msg");msg.textContent="";
  if(!m.date||!m.start||!m.end||!m.topic){msg.textContent="กรุณากรอกข้อมูลให้ครบ";return}
  if(m.end<=m.start){msg.textContent="เวลาจบต้องมากกว่าเวลาเริ่ม";return}
  if(!col){msg.textContent="ยังไม่ได้เชื่อมต่อฐานข้อมูล";return}
  try{await col.add(m);$("ft").value=""}catch(e){msg.textContent="บันทึกไม่ได้ (ไม่มีสิทธิ์แก้ไข)"}
};
$("up").onclick=async e=>{
  const id=e.target.dataset&&e.target.dataset.id;
  if(!id||!col||!confirm("ลบรายการนี้?"))return;
  try{await col.doc(id).delete()}catch(err){$("msg").textContent="ลบไม่ได้ (ไม่มีสิทธิ์แก้ไข)"}
};
render();setInterval(render,20000);
(async()=>{
  try{
    const db=await claude.use("db");
    if(!db){$("st").textContent="ยังไม่ได้เชื่อมต่อฐานข้อมูล";return}
    col=db.collection("meetings");
    col.onSnapshot(s=>{all=s.docs.map(d=>({id:d.id,...d.data()}));render()},()=>{$("st").textContent="โหลดข้อมูลไม่สำเร็จ"});
  }catch(e){$("st").textContent="ยังไม่ได้เชื่อมต่อฐานข้อมูล"}
})();
